class ResearchAgentNetwork:
    def __init__(self, query: str):
        self.search_agent = SearchAgent()
        self.scraper_agent = ScraperAgent()
        self.writer_agent = WriterAgent()
        self.critic_agent = CriticAgent()
    
    def execute(self):
        docs = self.search_agent.fetch(self.query)
        data = self.scraper_agent.extract(docs)
        draft = self.writer_agent.synthesize(data)
        return self.critic_agent.refine(draft)
